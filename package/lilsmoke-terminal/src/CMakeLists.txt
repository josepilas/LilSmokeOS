cmake_minimum_required(VERSION 3.16)
project(lilsmoke_terminal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VTERM REQUIRED IMPORTED_TARGET vterm>=0.3.3)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
include(PkgConfigPaths.cmake)
foreach(dependency VTERM SDL2 SDL2_TTF)
    lilsmoke_fix_pkg_target(PkgConfig::${dependency})
endforeach()
add_library(terminal-core STATIC Terminal.cpp Pty.cpp Keyboard.cpp)
target_link_libraries(terminal-core PUBLIC PkgConfig::VTERM PkgConfig::SDL2)
target_include_directories(terminal-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(terminal-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lilsmoke-terminal main.cpp)
target_compile_options(lilsmoke-terminal PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(lilsmoke-terminal PRIVATE terminal-core PkgConfig::SDL2_TTF)
install(TARGETS lilsmoke-terminal RUNTIME DESTINATION bin)
install(PROGRAMS lilsmoke-terminal-launch DESTINATION lib/lilsmoke)
option(LILSMOKE_TERMINAL_TESTS "Build isolated terminal tests" OFF)
if(LILSMOKE_TERMINAL_TESTS)
    enable_testing()
    add_executable(terminal-tests tests.cpp)
    target_link_libraries(terminal-tests PRIVATE terminal-core)
    target_compile_options(terminal-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    add_test(NAME terminal-core-pty-keyboard COMMAND terminal-tests)
    set_tests_properties(terminal-core-pty-keyboard PROPERTIES TIMEOUT 30)
endif()

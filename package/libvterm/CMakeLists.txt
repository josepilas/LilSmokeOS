# Build-only adapter for the unmodified pinned upstream C sources.
cmake_minimum_required(VERSION 3.16)
project(libvterm VERSION 0.3.3 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(vterm ${VTERM_SOURCES})
target_include_directories(vterm PUBLIC include)
set_target_properties(vterm PROPERTIES VERSION 0.0.0 SOVERSION 0)
set(INCDIR "${CMAKE_INSTALL_PREFIX}/include")
set(LIBDIR "${CMAKE_INSTALL_PREFIX}/lib")
set(VERSION "${PROJECT_VERSION}")
configure_file(vterm.pc.in vterm.pc @ONLY)
install(TARGETS vterm LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/vterm.h include/vterm_keycodes.h DESTINATION include)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vterm.pc" DESTINATION lib/pkgconfig)
